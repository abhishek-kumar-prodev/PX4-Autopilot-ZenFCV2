import os

# Extensions treated as text/code
TEXT_EXTENSIONS = {
    ".py",
    ".cpp",
    ".c",
    ".h",
    ".hpp",
    ".txt",
    ".md",
    ".json",
    ".yaml",
    ".yml",
    ".ini",
    ".cfg",
    ".sh",
    ".bash",
    ".js",
    ".ts",
    ".html",
    ".css",
}

IGNORE_DIRS = {
    ".git",
    "__pycache__",
    "build",
    "dist",
    ".vscode",
    ".idea",
    ".pytest_cache",
    "venv",
    ".venv",
}

IGNORE_FILES = {".DS_Store"}


def generate_tree(root_dir):
    tree_lines = []

    for root, dirs, files in os.walk(root_dir):
        dirs[:] = [d for d in dirs if d not in IGNORE_DIRS]

        level = root.replace(root_dir, "").count(os.sep)
        indent = "│   " * level
        folder_name = (
            os.path.basename(root) if level > 0 else os.path.basename(root_dir)
        )

        tree_lines.append(f"{indent}{folder_name}/")

        sub_indent = "│   " * (level + 1)
        for file in files:
            if file in IGNORE_FILES:
                continue
            tree_lines.append(f"{sub_indent}{file}")

    return "\n".join(tree_lines)


def is_text_file(filepath):
    ext = os.path.splitext(filepath)[1].lower()
    return ext in TEXT_EXTENSIONS


def dump_project(root_dir, output_file="project_dump.txt"):
    with open(output_file, "w", encoding="utf-8") as out:
        # Write tree structure
        out.write("# ==============================\n")
        out.write("# PROJECT STRUCTURE\n")
        out.write("# ==============================\n\n")
        out.write("```text\n")
        out.write(generate_tree(root_dir))
        out.write("\n```\n\n")

        # Write file contents
        out.write("# ==============================\n")
        out.write("# FILE CONTENTS\n")
        out.write("# ==============================\n")

        for root, dirs, files in os.walk(root_dir):
            dirs[:] = [d for d in dirs if d not in IGNORE_DIRS]

            for file in files:
                if file in IGNORE_FILES:
                    continue

                filepath = os.path.join(root, file)

                if not is_text_file(filepath):
                    continue

                rel_path = os.path.relpath(filepath, root_dir)

                try:
                    with open(filepath, "r", encoding="utf-8") as f:
                        content = f.read()
                except Exception:
                    continue

                language = os.path.splitext(file)[1][1:]

                out.write("\n\n# ==============================\n")
                out.write(f"# FILE: {rel_path}\n")
                out.write("# ==============================\n\n")
                out.write(f"```{language}\n")
                out.write(content)
                out.write("\n```\n")

    print(f"\n✅ Full project exported to {output_file}")


if __name__ == "__main__":
    root_directory = os.getcwd()
    dump_project(root_directory)

```

 * but since we want to be able to disable the NuttX use
 * of leds for system indication at will and there is no
 * separate switch, we need to build independent of the
 * CONFIG_ARCH_LEDS configuration switch.

__BEGIN_DECLS
extern void led_init(void);
extern void led_on(int led);
extern void led_off(int led);
extern void led_toggle(int led);

#  define xlat(p) (p)
static uint32_t g_ledmap[] = {
	GPIO_nLED_GREEN,   // Indexed by BOARD_LED_GREEN
	GPIO_nLED_BLUE,    // Indexed by BOARD_LED_BLUE
	GPIO_nLED_RED,     // Indexed by BOARD_LED_RED
};

__EXPORT void led_init(void)
{
	/* Configure LED GPIOs for output */
	for (size_t l = 0; l < (sizeof(g_ledmap) / sizeof(g_ledmap[0])); l++) {
		if (g_ledmap[l] != 0) {
			stm32_configgpio(g_ledmap[l]);
		}
	}
}

static void phy_set_led(int led, bool state)
{
	/* Drive Low to switch on */
	if (g_ledmap[led] != 0) {
		stm32_gpiowrite(g_ledmap[led], !state);
	}
}

static bool phy_get_led(int led)
{
	/* If Low it is on */
	if (g_ledmap[led] != 0) {
		return !stm32_gpioread(g_ledmap[led]);
	}

	return false;
}


__EXPORT void led_on(int led)
{
	phy_set_led(xlat(led), true);
}

__EXPORT void led_off(int led)
{
	phy_set_led(xlat(led), false);
}

__EXPORT void led_toggle(int led)
{
	phy_set_led(xlat(led), !phy_get_led(xlat(led)));
}

{
	bool present;

	present = !stm32_gpioread(GPIO_SDMMC1_NCD);

	if (sdio_dev && present != g_sd_inserted) {
		sdio_mediachange(sdio_dev, present);
		g_sd_inserted = present;
	}

	return OK;
}

/****************************************************************************
 * Name: stm32_sdio_initialize
 *
 * Description:
 *   Initialize SDIO-based MMC/SD card support
 *
 ****************************************************************************/

int stm32_sdio_initialize(void)
{
	int ret;

#ifdef HAVE_NCD
	/* Card detect */

	bool cd_status;

	/* Configure the card detect GPIO */

	stm32_configgpio(GPIO_SDMMC1_NCD);

	/* Register an interrupt handler for the card detect pin */

	stm32_gpiosetevent(GPIO_SDMMC1_NCD, true, true, true, stm32_ncd_interrupt);
#endif

	/* Mount the SDIO-based MMC/SD block driver */
	/* First, get an instance of the SDIO interface */

	finfo("Initializing SDIO slot %d\n", SDIO_SLOTNO);

	sdio_dev = sdio_initialize(SDIO_SLOTNO);

	if (!sdio_dev) {
		syslog(LOG_ERR, "[boot] Failed to initialize SDIO slot %d\n", SDIO_SLOTNO);
		return -ENODEV;
	}

	/* Now bind the SDIO interface to the MMC/SD driver */

	finfo("Bind SDIO to the MMC/SD driver, minor=%d\n", SDIO_MINOR);

	ret = mmcsd_slotinitialize(SDIO_MINOR, sdio_dev);

	if (ret != OK) {
		syslog(LOG_ERR, "[boot] Failed to bind SDIO to the MMC/SD driver: %d\n", ret);
		return ret;
	}

	finfo("Successfully bound SDIO to the MMC/SD driver\n");

#ifdef HAVE_NCD
	/* Use SD card detect pin to check if a card is g_sd_inserted */

	cd_status = !stm32_gpioread(GPIO_SDMMC1_NCD);
	finfo("Card detect : %d\n", cd_status);

	sdio_mediachange(sdio_dev, cd_status);
#else
	/* Assume that the SD card is inserted.  What choice do we have? */

	sdio_mediachange(sdio_dev, true);
#endif

	return OK;
}

#endif /* CONFIG_MMCSD */

	for (int i = 0; i < DIRECT_PWM_OUTPUT_CHANNELS; ++i) {
